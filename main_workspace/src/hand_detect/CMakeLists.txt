cmake_minimum_required(VERSION 3.18)
project(hand_detect LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(Threads REQUIRED)
find_package(CUDAToolkit REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0)
set(DEEPSTREAM_ROOT "/opt/nvidia/deepstream/deepstream" CACHE PATH "DeepStream SDK root")
find_path(DS_INCLUDE nvdsmeta.h PATHS "${DEEPSTREAM_ROOT}/sources/includes" REQUIRED)
find_library(DS_META nvds_meta PATHS "${DEEPSTREAM_ROOT}/lib" REQUIRED)
find_library(DS_GST_META nvdsgst_meta PATHS "${DEEPSTREAM_ROOT}/lib" REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/Inspection.msg" DEPENDENCIES std_msgs)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} "rosidl_typesupport_cpp")
ament_python_install_package(hand_detect_support PACKAGE_DIR python/hand_detect_support)

# Keep generated models, private configuration, logs, and temporary files in the repository.
get_filename_component(HAND_DETECT_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(HAND_DETECT_DATA_DIR "${HAND_DETECT_REPO_ROOT}/runtime/hand_detect")
file(MAKE_DIRECTORY "${HAND_DETECT_DATA_DIR}/logs/ros" "${HAND_DETECT_DATA_DIR}/tmp")
ament_environment_hooks("${CMAKE_CURRENT_SOURCE_DIR}/env/hand_detect.dsv.in")

add_library(hand_yolox_parser SHARED src/yolox_parser.cpp)
target_include_directories(hand_yolox_parser PRIVATE "${DS_INCLUDE}" ${CUDAToolkit_INCLUDE_DIRS})
target_compile_options(hand_yolox_parser PRIVATE -Wall -Wextra -Wpedantic)

add_executable(hand_detect_node src/hand_detect_node.cpp)
target_include_directories(hand_detect_node PRIVATE include "${DS_INCLUDE}")
ament_target_dependencies(hand_detect_node rclcpp std_msgs)
target_link_libraries(hand_detect_node "${typesupport}" PkgConfig::GST
  "${DS_META}" "${DS_GST_META}" Threads::Threads)
target_compile_options(hand_detect_node PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(hand_detect_node PROPERTIES INSTALL_RPATH "${DEEPSTREAM_ROOT}/lib")
install(TARGETS hand_detect_node DESTINATION lib/${PROJECT_NAME})
install(TARGETS hand_yolox_parser LIBRARY DESTINATION lib)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/setup_model.py DESTINATION lib/${PROJECT_NAME})
install(FILES README.md MODEL.md VALIDATION.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_result test/test_result.cpp)
  target_include_directories(test_result PRIVATE include)
  ament_add_gtest(test_parser test/test_parser.cpp)
  target_include_directories(test_parser PRIVATE "${DS_INCLUDE}" ${CUDAToolkit_INCLUDE_DIRS})
  target_link_libraries(test_parser hand_yolox_parser)
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
